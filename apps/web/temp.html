async function signMessage() { const doc = document.getElementById('document').value; const btn = document.querySelector('button'); btn.textContent = "Connecting to MCP..."; document.getElementById('results').style.display = 'none'; try { const response = await fetch('/api/sign', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer SYCRO_SECRET_TOKEN' }, body: JSON.stringify({ document: doc }) }); const data = await response.json(); if (data.success) { document.getElementById('res-hash').textContent = data.documentHash; document.getElementById('res-time').textContent = data.timestamp; document.getElementById('res-sig').textContent = data.signature; document.getElementById('results').style.display = 'block'; } else { alert("Error: " + data.error); } } catch (err) { alert("Connection error: " + err.message); } btn.textContent = "Sign Document"; }
